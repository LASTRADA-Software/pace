# pacectl — the operator's client for the cache and the compile fleet.
#
# Links the whole Pace library, the way pace-kv and compile-cache-testclient
# do, because it needs the library's own types rather than a copy of them: Cluster's
# state codec, FleetView's shared column table, and MetricsCatalog. That is the
# opposite choice from pace-cc, which deliberately links nothing so it can be
# spawned per translation unit with a static CRT — and it is the right one here,
# because this binary runs once when a person types a command.
#
# Linking Pace also brings the WIN32_LEAN_AND_MEAN / NOMINMAX / _WIN32_WINNT trio
# transitively (they are PUBLIC on that target), which is why — unlike the launcher —
# nothing here re-declares them.

# Everything except main.cpp, so the test binary can compile the same set.
#
# main.cpp is deliberately NOT in this list: it is in no test target, which is a
# property this tree has paid for twice (#370, #909), and it is why every decision
# this tool makes lives in one of the files below instead.
set(_pace_cli_sources
    CliAnswer.cpp
    CliCommand.cpp
    CliFormat.cpp
    CliValue.cpp
    CliVerbs.cpp
    DashboardLoop.cpp
    DashboardSampler.cpp
    FleetChartModel.cpp
    FleetFocus.cpp
    FleetDocument.cpp
    HistoryChart.cpp
    LiveEventSource.cpp
    LivePipedView.cpp
    LiveSession.cpp
    LiveStats.cpp
    LiveSubscriber.cpp
    MemcachedClient.cpp
    NodeClient.cpp
    NodeSlots.cpp
    NodeStatusText.cpp
    RespClient.cpp
    SocketExchange.cpp
    StatsGatherer.cpp
    StatsSource.cpp
    TerminalCapabilities.cpp
    DashboardGlyphs.cpp
    DashboardPanel.cpp
    DashboardPanels.cpp
    FleetReading.cpp
)

# The terminal half of the dashboard's input (#1372) is the one place this binary meets core-cpp's
# terminal UI, `core::tui`, so it is built only where that library is. With `PACE_BUILD_TUI` off,
# `MakeTerminalEvents` still links, and refuses by saying why, so a caller asking for a
# terminal gets a reason rather than a link error.
if(TARGET core::tui)
    list(APPEND _pace_cli_sources TerminalEventStream.cpp TerminalEvents.cpp)
    set(_pace_cli_tui_tests TerminalEventStream_test.cpp TerminalEvents_test.cpp)
else()
    list(APPEND _pace_cli_sources TerminalEventsUnavailable.cpp)
    set(_pace_cli_tui_tests "")
endif()

# The Sixel rung's encoder follows the same split: `TuiSixelEncoder.cpp` is the one file that
# includes `<core/tui/Sixel.hpp>`, and a build without it refuses by saying why.
if(TARGET core::tui)
    list(APPEND _pace_cli_sources TuiSixelEncoder.cpp)
    list(APPEND _pace_cli_tui_tests SixelEncoder_test.cpp)
else()
    list(APPEND _pace_cli_sources SixelEncoderUnavailable.cpp)
endif()

# How many cells text takes on a terminal is core-cpp's `core::tui::stringWidth`, reached through
# `TerminalCellWidth.cpp` alone. A build without the library has no terminal session to lay a frame
# out for, so its twin measures nothing: it names the build and ends the process.
if(TARGET core::tui)
    list(APPEND _pace_cli_sources TerminalCellWidth.cpp)
    list(APPEND _pace_cli_tui_tests TerminalCellWidth_test.cpp)
else()
    list(APPEND _pace_cli_sources TerminalCellWidthUnavailable.cpp)
endif()

add_executable(pacectl main.cpp ${_pace_cli_sources})
target_include_directories(pacectl PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(pacectl PRIVATE Pace)
if(TARGET core::tui)
    target_link_libraries(pacectl PRIVATE core::tui)
endif()

# The version is compiled in from the git tag rather than read at runtime, so
# `pacectl version` cannot report a value that has drifted from the build. The
# source refuses to compile without it.
target_compile_definitions(pacectl PRIVATE PACE_CLI_VERSION="${PACE_VERSION_STRING}")

install(TARGETS pacectl
    RUNTIME DESTINATION "${PACE_INSTALL_BINDIR}"
    COMPONENT "${PACE_APP_COMPONENT}"
)

# The command line, the verb handlers, the formatters and the stats ladder's choice are
# all pure functions over injected seams, so they are unit-testable with no daemon and
# no socket. Its own binary rather than rows in PaceTest: the library's test glob
# only reaches src/Pace.
if(PACE_BUILD_TESTS)
    add_executable(pacectl-tests
        ${_pace_cli_sources}
        CliAnswer_test.cpp
        CliCommand_test.cpp
        CliFormat_test.cpp
        CliValue_test.cpp
        CliVerbs_test.cpp
        DashboardLoop_test.cpp
        DashboardSampler_test.cpp
        FleetChartModel_test.cpp
        FleetFocus_test.cpp
        FleetDocument_test.cpp
        HistoryChart_test.cpp
        LiveEventSource_test.cpp
        LivePipedView_test.cpp
        LiveSession_test.cpp
        LiveStats_test.cpp
        LiveSubscriber_test.cpp
        MemcachedClient_test.cpp
        MemcachedVerbs_test.cpp
        NodeClient_test.cpp
        NodeSlots_test.cpp
        NodeVerbs_test.cpp
        RespClient_test.cpp
        StatsSource_test.cpp
        TerminalCapabilities_test.cpp
        DashboardGlyphs_test.cpp
        DashboardPanel_test.cpp
        FleetReading_test.cpp
        ${_pace_cli_tui_tests}
    )
    target_include_directories(pacectl-tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(pacectl-tests PRIVATE Pace Catch2::Catch2WithMain)
    if(TARGET core::tui)
        target_link_libraries(pacectl-tests PRIVATE core::tui)
    endif()
    target_compile_definitions(pacectl-tests PRIVATE PACE_CLI_VERSION="${PACE_VERSION_STRING}")
    include(Catch)
    # `SKIP_RETURN_CODE 4` tells ctest a Catch2 skip is a skip and not a failure
    # (#499), and under the Catch2 the top-level CMakeLists pins EXACTLY it is exact: a
    # run whose every case skipped exits 4 and any failure exits 42, whatever the
    # count. On 3.6.0 the exit code was the failure count, and a case failing four
    # assertions was scored *skipped* (#1128). `catch-skip-return-code` requires the
    # property, and `catch-skip-exit-contract` that it still means what it says.
    catch_discover_tests(pacectl-tests PROPERTIES SKIP_RETURN_CODE 4 ENVIRONMENT_MODIFICATION "${PACE_ERROR_DIALOG_ENVIRONMENT}")
endif()
